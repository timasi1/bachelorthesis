\section{Simplification of the hopping Hamiltonian $H_\text{hop.}$}
\label{chapter: AppendixA}

To further simplify the hopping Hamiltonian $H_\text{hop.}$, we begin by showing 
\begin{align}
    (\mathbf{d}_i \cdot\bm{\hat{\sigma}})^2 &= (d_x \hat{\sigma}_x + d_y \hat{\sigma}_y + d_z \hat{\sigma}_z)^2 \nonumber\\
    &= (d_x^2 + d_y^2 + d_z^2)\hat{I}_2 + d_xd_y\{\hat{\sigma}_x , \hat{\sigma}_y\} + d_yd_z\{\hat{\sigma}_y , \hat{\sigma}_z\} + d_xd_z\{\hat{\sigma}_x , \hat{\sigma}_z\} = \hat{I}_2
    \label{eq: appA d*sigma},
\end{align}
where $\hat{I}_2$ denotes the $2\times2$ identity matrix and the anticommutator is defined as $\{\hat{A} , \hat{B}\} = \hat{A}\hat{B}+\hat{B}\hat{A}$. Also, the unit-vector $|\mathbf{d}_i| = 1$ and $\{\hat{\sigma}_\mu, \hat{\sigma}_\nu\} = 2\delta_{\mu\nu}\hat{I}_2$ with $\mu,\nu\in\{x,y,z\}$ are used. 
Next, we can use Euler's formula and Eq.~\eqref{eq: appA d*sigma} to obtain
\begin{align}
    e^{i\Phi(\mathbf{d}_i\cdot\bm{\hat{\sigma}})} &= \sum_{j=0}^\infty \frac{(i\Phi)^{2j}}{(2j)!}\hat{I}_2 + \sum_{j=0}^\infty  \frac{(i\Phi)^{2j+1}}{(2j+1)!} (\mathbf{d}_i\cdot\bm{\hat{\sigma}}) = \cos(\Phi)\hat{I}_2 + i \sin(\Phi) (\mathbf{d}_i\cdot\bm{\hat{\sigma}})\nonumber\\
    &= \frac{1}{\sqrt{t^2+ \lambda^2}}\left\{t\hat{I}_2 - i\lambda (\bm{d}_i \cdot \bm{\hat{\sigma}}) \right\}.
    \label{eq: hamilton exp}
\end{align}

Here, we define $\cos(\Phi) = t/\sqrt{t^2 + \lambda^2}$ and $\sin(\Phi) = -\lambda/\sqrt{t^2 + \lambda^2}$ for $t^2+\lambda^2 >0$. The case for $t=\lambda=0$ is later fixed by defining $\hat{t}_i$, which vanishes in this case. Analogously, for the primed parameters the relation is given by
\begin{align}
    e^{i\Phi'(\bm{d}_i\cdot \bm{\hat{\sigma}})} = \frac{1}{\sqrt{t'^2 + \lambda'^2}} \left\{t'\hat{I}_2 - i\lambda' (\bm{d}_i \cdot \bm{\hat{\sigma}}) \right\},
    \label{eq: hamiltonian exp2}
\end{align}
where we use $\cos(\Phi') = t'/\sqrt{t'^2 + \lambda'^2}$ and $\sin(\Phi') = -\lambda'/\sqrt{t'^2 + \lambda'^2}$.
Thus, with Eqs.~\eqref{eq: hamilton exp} and \eqref{eq: hamiltonian exp2}, we obtain
\begin{align}
    H_{\text{hop.}}
&=
\sum_{\mathbf R}
\Bigg\{
\sum_{i=1}^{3}
\mathbf{c}_{\mathbf R,\beta_i}^{\dagger}
\left[t'\hat{I}_2 - i\lambda'(\bm{d}_i\!\cdot\!\bm{\hat{\sigma}})\right]
\mathbf{c}_{\mathbf R,\alpha_i}
\nonumber\\
& \qquad\qquad+
\sum_{i=1}^{3}
\mathbf{c}_{\mathbf R-\mathbf a_i,\beta_i}^{\dagger}
\left[t \hat{I}_2 - i\lambda(\bm{d}_i\!\cdot\!\bm{\hat{\sigma}})\right]
\mathbf{c}_{\mathbf R,\alpha_i}
\Bigg\}
+\mathrm{H.c.} \nonumber\\
&= 
\sum_{\mathbf R}
\Bigg\{
\sqrt{t'^2+\lambda'^2}
\sum_{i=1}^{3}
\mathbf{c}_{\mathbf R,\beta_i}^\dagger
e^{i\Phi'(\bm{d}_i\cdot\bm{\hat{\sigma}})}
\mathbf{c}_{\mathbf R,\alpha_i} \nonumber\\
& \qquad\qquad+
\sqrt{t^2+\lambda^2}
\sum_{i=1}^{3}
\mathbf{c}_{\mathbf R-\mathbf a_i,\beta_i}^\dagger
e^{i\Phi(\bm{d}_i\cdot\bm{\hat{\sigma}})}
\mathbf{c}_{\mathbf R,\alpha_i}
\Bigg\}
+\mathrm{H.c.}.
\label{eq: appendix hamiltonian}
\end{align}
This combines the Rashba-SOC term in Eq.~\eqref{eq:Rashba_Hamiltonian_soc} with the spin-independent hopping term in  Eq.~\eqref{eq: 1_hamiltonian_hopping}.
